Bu değerleri siz girersiniz, uygulama internetten çekmez. Doğru rakamları TÜİK/TCMB'den almanız gerekir. Eksik aylar %0 sayılır ve kıyas panelinde uyarı çıkar.

Fon detay penceresindeki "Fon Türü / Kategori / Varlık Dağılımı" bilgisi, genel CORS proxy'leri TEFAS'ın API'sini çoğu zaman engellediği için güvenilir çalışmıyor. Bunun yerine kendi Google Apps Script'inizi kullanmanız çok daha sağlam çalışır:
1) script.google.com'da yeni bir proje açın.
2) Aşağıdaki kodu yapıştırıp kaydedin, sonra Deploy → New deployment → Web app → "Execute as: Me", "Who has access: Anyone" seçip Deploy'a basın.
3) Verilen URL'yi buraya yapıştırın. (Daha önce dağıttıysanız: kodu bununla değiştirip Deploy → Manage deployments → Edit → New version ile yeniden dağıtın; BIST 100 kıyası için bu güncel kod gerekir.) 

function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  try {
    var p = e.parameter || {};
    // BIST 100 (ve diger Yahoo sembolleri) icin: ?symbol=XU100.IS&range=6mo
    if (p.symbol) {
      var range = (p.range || '6mo').toString();
      var url = 'https://query1.finance.yahoo.com/v8/finance/chart/' + encodeURIComponent(p.symbol) + '?range=' + range + '&interval=1d';
      var r = UrlFetchApp.fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0' }, muteHttpExceptions: true });
      if (r.getResponseCode() !== 200) return out({ ok: false, error: 'Yahoo HTTP ' + r.getResponseCode() });
      var res = JSON.parse(r.getContentText()).chart.result[0];
      var ts = res.timestamp, cl = res.indicators.quote[0].close, arr = [];
      for (var i = 0; ts.length > i; i++) {
        if (cl[i] != null) arr.push({ t: ts[i] * 1000, c: cl[i] });
      }
      return out({ ok: true, series: arr });
    }
    // TEFAS fon analizi icin: ?code=TGE
    var code = (p.code || '').toString().trim().toUpperCase();
    if (!code) return out({ ok: false, error: 'code veya symbol parametresi gerekli' });
    var resp = UrlFetchApp.fetch('https://www.tefas.gov.tr/api/DB/GetAllFundAnalyzeData', {
      method: 'post',
      contentType: 'application/x-www-form-urlencoded; charset=UTF-8',
      headers: { 'X-Requested-With': 'XMLHttpRequest', 'Referer': 'https://www.tefas.gov.tr/FonAnaliz.aspx', 'User-Agent': 'Mozilla/5.0' },
      payload: 'dil=TR&fonkod=' + encodeURIComponent(code),
      muteHttpExceptions: true
    });
    var status = resp.getResponseCode();
    if (status !== 200) return out({ ok: false, error: 'TEFAS HTTP ' + status });
    return out({ ok: true, data: JSON.parse(resp.getContentText('UTF-8')) });
  } catch (err) {
    return out({ ok: false, error: String(err) });
  }
}
